cmake_minimum_required(VERSION 3.16)
project(axiom_light_panel)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(std_msgs REQUIRED)
find_package(axiom_interfaces REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(${PROJECT_NAME} SHARED src/light_panel.cpp include/axiom_light_panel/light_panel.hpp)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
ament_target_dependencies(${PROJECT_NAME} rviz_common rclcpp pluginlib std_msgs axiom_interfaces)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  ament_add_gtest(test_light_panel test/test_light_panel.cpp)
  target_link_libraries(test_light_panel ${PROJECT_NAME} Qt5::Test)
  set_tests_properties(test_light_panel PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  # Manual opt-in acceptance: real RViz plugin + ROS + physical LED readback.
  # Deliberately not registered with CTest, because it changes connected hardware.
  add_executable(light_hardware_acceptance test/light_hardware_acceptance.cpp)
  target_link_libraries(light_hardware_acceptance ${PROJECT_NAME} Qt5::Test)
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(rviz_common rclcpp std_msgs axiom_interfaces)
ament_package()
